void fluffyMound(
    SceneGeometry& g,
    float x,
    float z,
    float radius,
    float height,
    int seed
) {
    g.dome({x, 0.015f, z}, radius, height, 8, 24);

    for (int i = 0; i < 6; ++i) {
        float a = 2.f * PI * i / 6.f + seed * 0.37f;
        float d = radius * (0.35f + (i % 3) * 0.08f);

        float px = x + std::cos(a) * d;
        float pz = z + std::sin(a) * d;

        g.dome(
            {px, 0.018f, pz},
            radius * (0.44f + (i % 3) * 0.07f),
            height * (0.48f + (i % 2) * 0.13f),
            6, 18
        );
    }
}
